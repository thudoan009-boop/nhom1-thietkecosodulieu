\section{Ánh Xạ Lược Đồ ER Phân Hệ Không Gian Làm Việc \& Tích Hợp VCS (Workspace \& External VCS Domain)}

\noindent\rule{\linewidth}{0.5pt}

\subsection{Cú pháp Rút gọn (Logical Schema Syntax)}
\begin{itemize}
    \item \texttt{workspaces (workspace\_id UUID PK, name VARCHAR(150), description TEXT, status VARCHAR(20), created\_at DATETIMEOFFSET, updated\_at DATETIMEOFFSET, deleted\_at DATETIMEOFFSET)}
    \item \texttt{github\_connections (connection\_id UUID PK, workspace\_id UUID FK, provider VARCHAR(50), account\_name VARCHAR(100), access\_token TEXT, sync\_status VARCHAR(20), created\_at DATETIME2, updated\_at DATETIME2)}
    \item \texttt{github\_connection\_scopes (scope\_id UUID PK, connection\_id UUID FK, scope\_value VARCHAR(100), created\_at DATETIME2)}
    \item \texttt{repositories (repository\_id UUID PK, workspace\_id UUID FK, name VARCHAR(150), url VARCHAR(255), default\_branch VARCHAR(50), created\_at DATETIME2, updated\_at DATETIME2)}
    \item \texttt{branches (branch\_id UUID PK, repository\_id UUID FK, name VARCHAR(100), is\_protected BOOLEAN, created\_at DATETIME2)}
\end{itemize}

\subsection{Mã DDL SQL Hoàn Chỉnh (\texttt{schema\_member\_workspace.sql})}

\begin{lstlisting}[language=SQL]
-- 1. BANG WORKSPACES (Quan ly khong gian lam viec cua nguoi dung)
CREATE TABLE workspaces (
    workspace_id UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    name VARCHAR(150) NOT NULL,
    description TEXT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
    created_at DATETIMEOFFSET DEFAULT SYSDATETIMEOFFSET(),
    updated_at DATETIMEOFFSET DEFAULT SYSDATETIMEOFFSET(),
    deleted_at DATETIMEOFFSET NULL,
    
    -- Constraints
    CONSTRAINT uk_workspace_id_name UNIQUE (workspace_id, name),
    CONSTRAINT chk_workspace_status CHECK (status IN ('ACTIVE', 'ARCHIVED'))
);

-- 2. BANG GITHUB_CONNECTIONS (Ket noi tai khoan GitHub voi Workspace)
CREATE TABLE github_connections (
    connection_id UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    workspace_id UNIQUEIDENTIFIER NOT NULL,
    provider VARCHAR(50) NOT NULL DEFAULT 'GITHUB',
    account_name VARCHAR(100) NOT NULL,
    access_token TEXT NOT NULL,
    sync_status VARCHAR(20) NOT NULL DEFAULT 'IDLE',
    created_at DATETIME2 NOT NULL DEFAULT GETDATE(),
    updated_at DATETIME2 NULL,

    -- Constraints
    CONSTRAINT uk_github_connection UNIQUE (connection_id, provider),
    CONSTRAINT chk_sync_status CHECK (sync_status IN ('IDLE', 'SYNCING', 'SUCCESS', 'FAILED'))
);

-- 3. BANG GITHUB_CONNECTION_SCOPES (Pham vi quyen han cua ket noi GitHub)
CREATE TABLE github_connection_scopes (
    scope_id UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    connection_id UNIQUEIDENTIFIER NOT NULL,
    scope_value VARCHAR(100) NOT NULL,
    created_at DATETIME2 NOT NULL DEFAULT GETDATE(),

    CONSTRAINT uk_connection_scope UNIQUE (connection_id, scope_value)
);

-- 4. BANG REPOSITORIES (Kho luu tru ma nguon lien ket)
CREATE TABLE repositories (
    repository_id UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    workspace_id UNIQUEIDENTIFIER NOT NULL,
    name VARCHAR(150) NOT NULL,
    url VARCHAR(255) NOT NULL,
    default_branch VARCHAR(50) DEFAULT 'main',
    created_at DATETIME2 NOT NULL DEFAULT GETDATE(),
    updated_at DATETIME2 NULL,

    CONSTRAINT uk_repository_name UNIQUE (repository_id, name)
);

-- 5. BANG BRANCHES (Cac nhanh lam viec trong Repository)
CREATE TABLE branches (
    branch_id UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    repository_id UNIQUEIDENTIFIER NOT NULL,
    name VARCHAR(100) NOT NULL,
    is_protected BIT NOT NULL DEFAULT 0,
    created_at DATETIME2 NOT NULL DEFAULT GETDATE(),

    CONSTRAINT uk_branch_name UNIQUE (repository_id, name)
);
GO

-- ============================================================
-- KHAI BAO CAC KHOA NGOAI THAM CHIEU (CROSS-DOMAIN & INTERNAL FKs)
-- ============================================================
ALTER TABLE github_connections ADD CONSTRAINT fk_github_connections_workspace 
    FOREIGN KEY (workspace_id) REFERENCES workspaces(workspace_id) ON DELETE CASCADE;

ALTER TABLE github_connection_scopes ADD CONSTRAINT fk_github_scopes_connection 
    FOREIGN KEY (connection_id) REFERENCES github_connections(connection_id) ON DELETE CASCADE;

ALTER TABLE repositories ADD CONSTRAINT fk_repositories_workspace 
    FOREIGN KEY (workspace_id) REFERENCES workspaces(workspace_id) ON DELETE CASCADE;

ALTER TABLE branches ADD CONSTRAINT fk_branches_repository 
    FOREIGN KEY (repository_id) REFERENCES repositories(repository_id) ON DELETE CASCADE;
GO
\end{lstlisting}
