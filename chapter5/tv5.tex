\chapter{Các Truy vấn Hệ thống Demo}

Dưới đây là danh sách các truy vấn SQL minh họa cho các tính năng của hệ thống.

\section{Truy vấn Phân quyền (RBAC): Danh sách Người dùng và Vai trò trong Workspace}
Hiển thị nhân sự đang tham gia vào các không gian làm việc và chức danh của họ.

\begin{lstlisting}[language=SQL]
SELECT 
    w.name AS WorkspaceName,
    u.first_name + ' ' + u.last_name AS FullName,
    u.email,
    r.role_name AS Role,
    wm.status AS MemberStatus
FROM workspaces w
JOIN workspace_members wm ON w.workspace_id = wm.workspace_id
JOIN users u ON wm.user_id = u.user_id
JOIN roles r ON wm.role_id = r.role_id
ORDER BY w.name, r.role_name;
\end{lstlisting}

\section{Truy vấn Phân quyền: Kiểm tra chi tiết Quyền hạn (Permissions) của một Vai trò}
Demo cách hệ thống biết Developer hay Manager được phép làm gì.

\begin{lstlisting}[language=SQL]
SELECT 
    r.role_name,
    p.permission_code,
    p.description
FROM roles r
JOIN role_permissions rp ON r.role_id = rp.role_id
JOIN permissions p ON rp.permission_id = p.permission_id
WHERE r.role_name = 'developer'
ORDER BY p.permission_code;
\end{lstlisting}

\section{Truy vấn Quản lý Repo: Thống kê số lượng Nhánh (Branch) theo từng Repository}
Hiển thị tổng quan các kho lưu trữ mã nguồn.

\begin{lstlisting}[language=SQL]
SELECT 
    r.name AS RepositoryName,
    r.default_branch,
    COUNT(b.branch_id) AS TotalBranches
FROM repositories r
LEFT JOIN branches b ON r.repository_id = b.repository_id
GROUP BY r.name, r.default_branch;
\end{lstlisting}

\section{Truy vấn Git Workflow: Chi tiết các Commit nằm trong một Pull Request}
Sử dụng bảng trung gian \texttt{pull\_request\_commits} để truy xuất lịch sử đẩy code.

\begin{lstlisting}[language=SQL]
SELECT 
    pr.pr_number,
    pr.title AS PR_Title,
    c.commit_hash,
    c.author_name,
    c.message AS CommitMessage,
    c.committed_at
FROM pull_requests pr
JOIN pull_request_commits prc ON pr.pr_id = prc.pr_id
JOIN commits c ON prc.commit_id = c.commit_id
WHERE pr.status = 'MERGED'
ORDER BY pr.pr_number, c.committed_at DESC;
\end{lstlisting}

\section{Truy vấn Git Workflow: Phân tích Dòng code thay đổi (Code Diff)}
Demo việc hệ thống lưu trữ sự khác biệt mã nguồn khi lập trình viên sửa file.

\begin{lstlisting}[language=SQL]
SELECT 
    c.commit_hash,
    f.file_path,
    cd.change_type,
    cd.additions,
    cd.deletions,
    cd.diff_content
FROM commits c
JOIN code_diffs cd ON c.commit_id = cd.commit_id
JOIN files f ON cd.file_id = f.file_id
WHERE cd.change_type = 'MODIFIED';
\end{lstlisting}

\section{Truy vấn Cây cú pháp (AST): Danh sách Thực thể mã nguồn (Code Entities)}
Hiển thị các hàm, lớp, biến được hệ thống bóc tách từ file source code.

\begin{lstlisting}[language=SQL]
SELECT 
    f.file_path,
    ce.entity_name,
    ce.entity_type,
    ce.signature,
    ce.start_line,
    ce.end_line
FROM files f
JOIN code_entities ce ON f.file_id = ce.file_id
ORDER BY f.file_path, ce.start_line;
\end{lstlisting}

\section{Truy vấn Đệ quy AST: Sự phụ thuộc giữa các đoạn code (Dependencies)}
Tìm xem hàm A đang gọi đến hoặc phụ thuộc vào hàm B nào.

\begin{lstlisting}[language=SQL]
SELECT 
    src.entity_name AS SourceEntity,
    src.entity_type AS SourceType,
    d.dependency_type,
    tgt.entity_name AS TargetEntity,
    tgt.entity_type AS TargetType
FROM dependencies d
JOIN code_entities src ON d.source_entity_id = src.code_entity_id
JOIN code_entities tgt ON d.target_entity_id = tgt.code_entity_id
WHERE d.dependency_type = 'REFERENCES';
\end{lstlisting}

\section{Truy vấn Sinh tài liệu: Cấu trúc của một Biểu mẫu (Template)}
Demo các phân đoạn và điểm giữ chỗ (Placeholder) được định nghĩa bởi Manager.

\begin{lstlisting}[language=SQL]
SELECT 
    t.name AS TemplateName,
    ts.order_index,
    ts.heading_title,
    p.variable_name,
    p.expected_data_type
FROM templates t
JOIN template_sections ts ON t.template_id = ts.template_id
LEFT JOIN placeholders p ON ts.section_id = p.section_id
ORDER BY t.name, ts.order_index;
\end{lstlisting}

\section{Truy vấn Automation: Cách Placeholder map với Source Code}
Chứng minh tính tự động hóa: Điểm điền tài liệu nào đang map với hàm code nào.

\begin{lstlisting}[language=SQL]
SELECT 
    p.variable_name AS Placeholder,
    tm.extraction_rule,
    ce.entity_name AS MappedCodeEntity,
    ce.signature AS CodeSignature
FROM template_mappings tm
JOIN placeholders p ON tm.placeholder_id = p.placeholder_id
JOIN code_entities ce ON tm.code_entity_id = ce.code_entity_id
WHERE tm.is_active = 1;
\end{lstlisting}

\section{Truy vấn Vòng đời Tài liệu: Các Phiên bản của một Tài liệu}
Xem lịch sử cập nhật version của một file tài liệu.

\begin{lstlisting}[language=SQL]
SELECT 
    d.title AS DocumentTitle,
    d.document_type,
    dv.version_number,
    dv.status AS VersionStatus,
    dv.created_at
FROM documents d
JOIN document_versions dv ON d.document_id = dv.document_id
ORDER BY d.title, dv.created_at DESC;
\end{lstlisting}

\section{Truy vấn Review: Đọc các Comment góp ý trên Tài liệu}
Kiểm tra xem Technical Lead hoặc Manager đã nhận xét gì về tài liệu.

\begin{lstlisting}[language=SQL]
SELECT 
    dv.version_number,
    u.first_name + ' ' + u.last_name AS Reviewer,
    r.decision AS ReviewDecision,
    rc.content AS CommentContent,
    rc.is_resolved
FROM document_versions dv
JOIN reviews r ON dv.version_id = r.version_id
JOIN review_comments rc ON r.review_id = rc.review_id
JOIN users u ON rc.user_id = u.user_id;
\end{lstlisting}

\section{Truy vấn Workflow Phê duyệt: Lịch sử Approval}
Truy xuất ai đã chốt duyệt phiên bản tài liệu cuối cùng.

\begin{lstlisting}[language=SQL]
SELECT 
    dv.version_number,
    u.first_name + ' ' + u.last_name AS Approver,
    a.decision,
    a.rejection_reason,
    a.approved_at
FROM document_versions dv
JOIN approvals a ON dv.version_id = a.version_id
JOIN users u ON a.user_id = u.user_id;
\end{lstlisting}

\section{Truy vấn Grounding: Dẫn chứng mã nguồn của Tài liệu}
Chứng minh tài liệu không bịa đặt (hallucination) nhờ bằng chứng gốc.

\begin{lstlisting}[language=SQL]
SELECT 
    dv.version_number,
    ge.purpose,
    ge.snippet_content,
    ce.entity_name AS SourceCodeEntity,
    f.file_path
FROM document_versions dv
JOIN grounding_evidences ge ON dv.version_id = ge.version_id
LEFT JOIN grounding_evidence_code_entities gece ON ge.evidence_id = gece.evidence_id
LEFT JOIN code_entities ce ON gece.code_entity_id = ce.code_entity_id
LEFT JOIN files f ON ce.file_id = f.file_id;
\end{lstlisting}

\section{Truy vấn Cảnh báo Trôi dạt (Drift Alert): Các sai lệch chưa xử lý}
Hiển thị các cảnh báo khi code mới sửa làm tài liệu cũ bị sai lệch.

\begin{lstlisting}[language=SQL]
SELECT 
    r.name AS Repository,
    da.drift_type,
    da.status AS AlertStatus,
    da.detected_at,
    ge.snippet_content AS OldDocumentedCode
FROM drift_alerts da
JOIN repositories r ON da.repository_id = r.repository_id
JOIN grounding_evidences ge ON da.evidence_id = ge.evidence_id
WHERE da.status = 'OPEN'
ORDER BY da.detected_at DESC;
\end{lstlisting}

\section{Truy vấn Audit: Nhật ký thay đổi hệ thống (Change Log)}
Truy xuất toàn bộ lịch sử biến động để audit hệ thống.

\begin{lstlisting}[language=SQL]
SELECT 
    dv.version_number,
    cl.change_description,
    cl.diff_content,
    cl.created_at
FROM change_logs cl
JOIN document_versions dv ON cl.version_id = dv.version_id
ORDER BY cl.created_at DESC;
\end{lstlisting}
