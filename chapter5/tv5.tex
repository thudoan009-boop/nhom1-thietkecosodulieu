\section{Fill Data}
\label{sec:fill-data}

Sau khi hoàn thành việc xây dựng lược đồ vật lý, dữ liệu mẫu được nạp vào
các bảng nhằm kiểm tra khả năng hoạt động và tính nhất quán của cơ sở dữ liệu.
Script Fill Data sử dụng các bảng tạm để sinh và ánh xạ UUID giữa các thực thể,
sau đó thực hiện chèn dữ liệu theo đúng các mối quan hệ khóa chính và khóa ngoại.
Việc nạp dữ liệu giúp kiểm thử cấu trúc cơ sở dữ liệu và tạo cơ sở cho các bước
truy vấn, kiểm tra và đánh giá hệ thống ở các giai đoạn tiếp theo.

DECLARE @i INT = 1;
WHILE @i <= 60 
BEGIN
    IF @i <= 12 INSERT INTO #Users VALUES (@i, NEWID());
    IF @i <= 5  INSERT INTO #Roles VALUES (@i, NEWID());
    IF @i <= 15 INSERT INTO #Perms VALUES (@i, NEWID());
    IF @i <= 35 INSERT INTO #RolePerms VALUES (@i, NEWID());
    IF @i <= 3  INSERT INTO #Workspaces VALUES (@i, NEWID());
    IF @i <= 20 INSERT INTO #WsMembers VALUES (@i, NEWID());
    IF @i <= 4  INSERT INTO #GitConns VALUES (@i, NEWID());
    IF @i <= 12 INSERT INTO #GitScopes VALUES (@i, NEWID());
    IF @i <= 5  INSERT INTO #Repos VALUES (@i, NEWID());
    IF @i <= 12 INSERT INTO #Branches VALUES (@i, NEWID());
    IF @i <= 8  INSERT INTO #PRs VALUES (@i, NEWID());
    IF @i <= 15 INSERT INTO #PRLabels VALUES (@i, NEWID());
    IF @i <= 30 INSERT INTO #Commits VALUES (@i, NEWID());
    IF @i <= 40 INSERT INTO #Files VALUES (@i, NEWID());
    IF @i <= 15 INSERT INTO #CodeDiffs VALUES (@i, NEWID());
    IF @i <= 50 INSERT INTO #Entities VALUES (@i, NEWID());
    IF @i <= 60 INSERT INTO #Deps VALUES (@i, NEWID());
    IF @i <= 4  INSERT INTO #Templates VALUES (@i, NEWID());
    IF @i <= 20 INSERT INTO #Sections VALUES (@i, NEWID());
    IF @i <= 20 INSERT INTO #Placeholders VALUES (@i, NEWID());
    IF @i <= 30 INSERT INTO #Mappings VALUES (@i, NEWID());
    IF @i <= 8  INSERT INTO #Docs VALUES (@i, NEWID());
    IF @i <= 20 INSERT INTO #DocVers VALUES (@i, NEWID());
    IF @i <= 30 INSERT INTO #Evidences VALUES (@i, NEWID());
    IF @i <= 12 INSERT INTO #Reviews VALUES (@i, NEWID());
    IF @i <= 25 INSERT INTO #ReviewCmts VALUES (@i, NEWID());
    IF @i <= 8  INSERT INTO #Approvals VALUES (@i, NEWID());
    IF @i <= 30 INSERT INTO #ChangeLogs VALUES (@i, NEWID());
    IF @i <= 8  INSERT INTO #Drifts VALUES (@i, NEWID());
    IF @i <= 12 INSERT INTO #Notifs VALUES (@i, NEWID());
    
    -- 5 bảng mới thêm
    IF @i <= 40 INSERT INTO #CommitFiles VALUES (@i, NEWID());
    IF @i <= 35 INSERT INTO #EvidenceEntities VALUES (@i, NEWID());
    IF @i <= 38 INSERT INTO #BranchCommits VALUES (@i, NEWID());
    IF @i <= 24 INSERT INTO #PRCommits VALUES (@i, NEWID());
    IF @i <= 28 INSERT INTO #CommitParents VALUES (@i, NEWID());
    
    SET @i = @i + 1;
END
