\subsection*{Lược đồ Dạng Cú pháp Rút gọn}
\begin{itemize}
    \item \textbf{repositories} (repository\_id UUID PK, workspace\_id UUID FK, name VARCHAR(255), url VARCHAR(255) UNIQUE, default\_branch VARCHAR(100), sync\_status VARCHAR(20), last\_synced\_at TIMESTAMP, created\_at TIMESTAMP, updated\_at TIMESTAMP) 
    \item \textbf{branches} (branch\_id UUID PK, repository\_id UUID FK, branch\_name VARCHAR(100), is\_default BOOLEAN, created\_at TIMESTAMP, updated\_at TIMESTAMP) 
    \item \textbf{commits} (commit\_id UUID PK, branch\_id UUID FK, user\_id UUID FK, commit\_hash VARCHAR(40) UNIQUE, message TEXT, author\_name VARCHAR(100), author\_email VARCHAR(255), committed\_at TIMESTAMP, created\_at TIMESTAMP) 
    \item \textbf{pull\_requests} (pr\_id UUID PK, repository\_id UUID FK, pr\_number INT, title VARCHAR(255), description TEXT, source\_branch VARCHAR(100), target\_branch VARCHAR(100), status VARCHAR(20), created\_at TIMESTAMP, updated\_at TIMESTAMP, merged\_at TIMESTAMP) 
    \item \textbf{pull\_request\_labels} (label\_id UUID PK, pr\_id UUID FK, label\_name VARCHAR(50)) 
    \item \textbf{code\_diffs} (diff\_id UUID PK, commit\_id UUID FK, pr\_id UUID FK, file\_path VARCHAR(500), change\_type VARCHAR(20), additions INT, deletions INT, diff\_content TEXT, created\_at TIMESTAMP) 
    \item \textbf{files} (file\_id UUID PK, repository\_id UUID FK, file\_path VARCHAR(500), file\_type VARCHAR(20), status VARCHAR(20), created\_at TIMESTAMP, updated\_at TIMESTAMP) 
    \item \textbf{code\_entities} (entity\_id UUID PK, file\_id UUID FK, entity\_name VARCHAR(100), entity\_type VARCHAR(50), signature TEXT, doc\_comment TEXT, start\_line INT, end\_line INT, created\_at TIMESTAMP, updated\_at TIMESTAMP) 
    \item \textbf{dependencies} (dependency\_id UUID PK, source\_entity\_id UUID FK, target\_entity\_id UUID FK, dependency\_type VARCHAR(50), created\_at TIMESTAMP)
\end{itemize}
