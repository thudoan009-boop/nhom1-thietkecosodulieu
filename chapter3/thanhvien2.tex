\subsection{Lược đồ Dạng Cú pháp Rút gọn}
\begin{itemize}
    \item \textbf{repositories} (repository\_id UUID PK, workspace\_id UUID FK, name, url UNIQUE, default\_branch, sync\_status, last\_synced\_at, created\_at, updated\_at) 
    \item \textbf{branches} (branch\_id UUID PK, repository\_id UUID FK, branch\_name, is\_default, created\_at, updated\_at) 
    \item \textbf{commits} (commit\_id UUID PK, branch\_id UUID FK, user\_id UUID FK, commit\_hash UNIQUE, message, author\_name, author\_email, committed\_at, created\_at) 
    \item \textbf{pull\_requests} (pr\_id UUID PK, repository\_id UUID FK, pr\_number, title, description, source\_branch, target\_branch, status, created\_at, updated\_at, merged\_at) 
    \item \textbf{pull\_request\_labels} (label\_id UUID PK, pr\_id UUID FK, label\_name) 
    \item \textbf{code\_diffs} (diff\_id UUID PK, commit\_id UUID FK, pr\_id UUID FK, file\_path, change\_type, additions, deletions, diff\_content, created\_at) 
    \item \textbf{files} (file\_id UUID PK, repository\_id UUID FK, file\_path, file\_type, status, created\_at, updated\_at) 
    \item \textbf{code\_entities} (entity\_id UUID PK, file\_id UUID FK, entity\_name, entity\_type, signature, doc\_comment, start\_line, end\_line, created\_at, updated\_at) 
    \item \textbf{dependencies} (dependency\_id UUID PK, source\_entity\_id UUID FK, target\_entity\_id UUID FK, dependency\_type, created\_at)
\end{itemize}

\subsection{Ràng buộc Tham chiếu (Foreign Key Constraints)}
\begin{itemize}
    \item \textbf{FK\_repositories\_workspace:} repositories(workspace\_id) $\rightarrow$ workspaces(workspace\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_branches\_repositories:} branches(repository\_id) $\rightarrow$ repositories(repository\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_commits\_branches:} commits(branch\_id) $\rightarrow$ branches(branch\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_commits\_users:} commits(user\_id) $\rightarrow$ users(user\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE SET NULL / ON UPDATE CASCADE
    \item \textbf{FK\_pull\_requests\_repositories:} pull\_requests(repository\_id) $\rightarrow$ repositories(repository\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_pr\_labels\_pr:} pull\_request\_labels(pr\_id) $\rightarrow$ pull\_requests(pr\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_code\_diffs\_commits:} code\_diffs(commit\_id) $\rightarrow$ commits(commit\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_code\_diffs\_pr:} code\_diffs(pr\_id) $\rightarrow$ pull\_requests(pr\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_files\_repositories:} files(repository\_id) $\rightarrow$ repositories(repository\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_code\_entities\_files:} code\_entities(file\_id) $\rightarrow$ files(file\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_dependencies\_source:} dependencies(source\_entity\_id) $\rightarrow$ code\_entities(entity\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
    \item \textbf{FK\_dependencies\_target:} dependencies(target\_entity\_id) $\rightarrow$ code\_entities(entity\_id) \newline $\Rightarrow$ Hành vi tham chiếu: ON DELETE CASCADE / ON UPDATE CASCADE
\end{itemize}

\subsection{Ràng buộc Giá trị Miền (CHECK Constraints)}
\begin{itemize}
    \item \textbf{CHK\_repo\_sync\_status:} CHECK (sync\_status IN ('SYNCING', 'SUCCESS', 'FAILED'))
    \item \textbf{CHK\_pr\_status:} CHECK (status IN ('OPEN', 'MERGED', 'CLOSED'))
    \item \textbf{CHK\_diff\_change\_type:} CHECK (change\_type IN ('ADD', 'MODIFY', 'DELETE', 'RENAME'))
    \item \textbf{CHK\_file\_status:} CHECK (status IN ('ACTIVE', 'DELETED', 'RENAMED'))
    \item \textbf{CHK\_entity\_type:} CHECK (entity\_type IN ('CLASS', 'METHOD', 'FUNCTION', 'INTERFACE', 'ENDPOINT'))
    \item \textbf{CHK\_dependency\_type:} CHECK (dependency\_type IN ('CALLS', 'INHERITS', 'IMPLEMENTS', 'USES', 'IMPORTS'))
\end{itemize}

\subsection{Báo cáo Chứng minh Chuẩn hóa 3NF}

\subsubsection{Khai báo Tập Phụ thuộc Hàm (Functional Dependencies - FDs)}
Xét bảng \textbf{repositories} có tập phụ thuộc hàm $F = \{f_1, f_2\}$ như sau:
\begin{itemize}
    \item $f_1$: repository\_id $\rightarrow$ workspace\_id, name, url, default\_branch, sync\_status, last\_synced\_at, created\_at, updated\_at
    \item $f_2$: url $\rightarrow$ repository\_id (Khóa ứng viên duy nhất định danh đường dẫn kho chứa trên VCS)
\end{itemize}

Xét bảng \textbf{commits} có tập phụ thuộc hàm:
\begin{itemize}
    \item $f_3$: commit\_id $\rightarrow$ branch\_id, user\_id, commit\_hash, message, author\_name, author\_email, committed\_at, created\_at
    \item $f_4$: commit\_hash $\rightarrow$ commit\_id (Khóa ứng viên duy nhất từ Git Hash)
\end{itemize}

\subsubsection{Đánh giá Cấp Chuẩn}
\begin{enumerate}
    \item \textbf{Đạt Chuẩn 1NF (First Normal Form):}
    \begin{itemize}
        \item Tất cả các cột trong các bảng của phân hệ đều chứa giá trị nguyên tố (Atomic Value).
        \item Thuộc tính phức hợp \textit{AuthorInfo} (của Commit) đã được phẳng hóa thành \texttt{author\_name} và \texttt{author\_email}.
        \item Thuộc tính phức hợp \textit{position} (của CodeEntity) đã được phẳng hóa thành \texttt{start\_line} và \texttt{end\_line}.
        \item Thuộc tính đa trị \textit{labels} (của PullRequest) đã được tách triệt để sang bảng phụ \texttt{pull\_request\_labels}.
    \end{itemize}
    
    \item \textbf{Đạt Chuẩn 2NF (Second Normal Form):}
    \begin{itemize}
        \item Các bảng trong lược đồ đều sử dụng Khóa chính đơn (Surrogate PK dạng UUID như \texttt{repository\_id}, \texttt{commit\_id}, \texttt{pr\_id},...), không sử dụng khóa chính phức hợp cho các thực thể gốc.
        \item Do đó, mọi thuộc tính không khóa đều phụ thuộc hoàn toàn và trực tiếp vào toàn bộ khóa chính, hoàn toàn không tồn tại phụ thuộc một phần (Partial Dependency).
    \end{itemize}
    
    \item \textbf{Đạt Chuẩn 3NF (Third Normal Form):}
    \begin{itemize}
        \item Mọi thuộc tính không khóa trong các bảng đều phụ thuộc trực tiếp vào Khóa chính (ví dụ trong bảng \texttt{commits}, các thông tin như \texttt{message}, \texttt{author\_name}, \texttt{committed\_at} phụ thuộc trực tiếp vào \texttt{commit\_id}).
        \item Không tồn tại phụ thuộc bắc cầu $X \rightarrow Y \rightarrow Z$ giữa các thuộc tính không khóa.
    \end{itemize}
\end{enumerate}

\noindent\textbf{Kết luận:} Lược đồ quan hệ phân hệ VCS \& Code Structure đạt tiêu chuẩn \textbf{3NF}, đảm bảo tính toàn vẹn dữ liệu, loại bỏ các dị thường (anomalies) khi thêm, sửa, xóa và tối ưu hóa hiệu suất lưu trữ cho hệ thống.
