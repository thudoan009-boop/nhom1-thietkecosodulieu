\subsection{Lược đồ dạng cú pháp rút gọn}

\begin{itemize}
    \item \texttt{templates(template\_id PK, workspace\_id FK, name, description, content\_structure, creator\_type, status, created\_at, updated\_at)}

    \item \texttt{template\_sections(section\_id PK, template\_id FK, heading\_title, order\_index, prompt\_instruction, created\_at, updated\_at, UNIQUE(template\_id, order\_index))}

    \item \texttt{placeholders(placeholder\_id PK, section\_id FK, variable\_name, expected\_data\_type, description, is\_required, created\_at, updated\_at, UNIQUE(section\_id, variable\_name))}

    \item \texttt{template\_mappings(mapping\_id PK, placeholder\_id FK, entity\_id FK, extraction\_rule, is\_active, UNIQUE(placeholder\_id, entity\_id))}

    \item \texttt{documents(document\_id PK, workspace\_id FK, user\_id FK, title, document\_type, status, created\_at, updated\_at, deleted\_at)}

    \item \texttt{document\_versions(version\_id PK, document\_id FK, version\_number, content, change\_summary, status, ticketed\_codes, created\_at, updated\_at, UNIQUE(document\_id, version\_number))}

    \item \texttt{change\_logs(log\_id PK, version\_id FK, change\_type, change\_description, diff\_content, created\_at, updated\_at)}
\end{itemize}


\subsection{Báo cáo Chứng minh Chuẩn hóa 3NF}

\subsubsection{Khai báo Tập Phụ thuộc Hàm 
(Functional Dependencies -- FDs)}

Xét bảng \texttt{templates} có tập phụ thuộc hàm như sau:

\[
\texttt{template\_id}
\rightarrow
\{
\texttt{workspace\_id},
\texttt{name},
\texttt{description},
\texttt{content\_structure},
\texttt{creator\_type},
\texttt{status},
\texttt{created\_at},
\texttt{updated\_at}
\}
\]

Xét bảng \texttt{template\_sections} có tập phụ thuộc hàm:

\[
\texttt{section\_id}
\rightarrow
\{
\texttt{template\_id},
\texttt{heading\_title},
\texttt{order\_index},
\texttt{prompt\_instruction},
\texttt{created\_at},
\texttt{updated\_at}
\}
\]

Ngoài ra:

\[
(\texttt{template\_id},\texttt{order\_index})
\rightarrow
\texttt{section\_id}
\]

vì có ràng buộc
\texttt{UNIQUE(template\_id, order\_index)}.

\subsubsection{Đánh giá Cấp Chuẩn}

\paragraph{1. Đạt Chuẩn 1NF (First Normal Form)}

\begin{itemize}
    \item Tất cả các cột trong các bảng của phân hệ đều chứa giá trị nguyên tố (Atomic Value).

    \item Các thuộc tính được lưu trữ dưới dạng đơn trị, không có nhóm thuộc tính lặp hay thuộc tính đa trị trong cùng một ô.

    \item Các thông tin có tính chất riêng biệt đã được tách thành các bảng tương ứng. Ví dụ, thông tin về Section được lưu trong \texttt{template\_sections}, thông tin Placeholder được lưu trong \texttt{placeholders}, còn thông tin phiên bản tài liệu được lưu trong \texttt{document\_versions}.
\end{itemize}


\paragraph{2. Đạt Chuẩn 2NF (Second Normal Form)}

\begin{itemize}
    \item Các bảng trong lược đồ đều sử dụng Khóa chính đơn (Surrogate PK) như \texttt{template\_id}, \texttt{section\_id}, \texttt{placeholder\_id}, \texttt{mapping\_id}, \texttt{document\_id}, \texttt{version\_id}, \texttt{log\_id}, không sử dụng khóa chính phức hợp cho các bảng.

    \item Do khóa chính của các bảng chỉ gồm một thuộc tính, không tồn tại trường hợp thuộc tính không khóa chỉ phụ thuộc vào một phần của khóa chính.

    \item Vì vậy, không tồn tại phụ thuộc bộ phận (Partial Dependency).

    \item Các ràng buộc UNIQUE như \texttt{UNIQUE(template\_id, order\_index)}, \texttt{UNIQUE(section\_id, variable\_name)}, \texttt{UNIQUE(placeholder\_id, entity\_id)} và \texttt{UNIQUE(document\_id, version\_number)} được sử dụng để đảm bảo tính duy nhất của dữ liệu, nhưng không làm phát sinh phụ thuộc bộ phận đối với khóa chính đơn.
\end{itemize}


\paragraph{3. Đạt Chuẩn 3NF (Third Normal Form)}

\begin{itemize}
    \item Mọi thuộc tính không khóa trong các bảng đều phụ thuộc trực tiếp vào Khóa chính của bảng.

    \item Ví dụ trong bảng \texttt{templates}, các thông tin như \texttt{workspace\_id}, \texttt{name}, \texttt{description}, \texttt{content\_structure}, \texttt{creator\_type}, \texttt{status}, \texttt{created\_at}, \texttt{updated\_at} đều phụ thuộc trực tiếp vào \texttt{template\_id}.

    \item Trong bảng \texttt{document\_versions}, các thông tin như \texttt{document\_id}, \texttt{version\_number}, \texttt{content}, \texttt{change\_summary}, \texttt{status}, \texttt{ticketed\_codes}, \texttt{created\_at}, \texttt{updated\_at} phụ thuộc trực tiếp vào \texttt{version\_id}.

    \item Tương tự, trong bảng \texttt{change\_logs}, các thuộc tính \texttt{version\_id}, \texttt{change\_type}, \texttt{change\_description}, \texttt{diff\_content}, \texttt{created\_at}, \texttt{updated\_at} phụ thuộc trực tiếp vào \texttt{log\_id}.

    \item Không tồn tại phụ thuộc bắc cầu (Transitive Dependency) giữa các thuộc tính không khóa trong phạm vi các phụ thuộc hàm được xác định.
\end{itemize}
