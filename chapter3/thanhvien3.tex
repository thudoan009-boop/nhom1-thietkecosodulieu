\section{Lược đồ dạng cú pháp rút gọn}

\begin{itemize}

    \item \texttt{templates(
    template\_id UUID PK NOT NULL,
    workspace\_id UUID FK NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT NULL,
    content\_structure TEXT NOT NULL,
    creator\_type VARCHAR(50) NOT NULL,
    status VARCHAR(50) NOT NULL,
    created\_at TIMESTAMP NOT NULL,
    updated\_at TIMESTAMP NOT NULL)}

    \item \texttt{template\_sections(
    section\_id UUID PK NOT NULL,
    template\_id UUID FK NOT NULL,
    heading\_title VARCHAR(255) NOT NULL,
    order\_index INT NOT NULL,
    prompt\_instruction TEXT NULL,
    created\_at TIMESTAMP NOT NULL,
    updated\_at TIMESTAMP NOT NULL,
    UNIQUE(template\_id, order\_index))}

    \item \texttt{placeholders(
    placeholder\_id UUID PK NOT NULL,
    section\_id UUID FK NOT NULL,
    variable\_name VARCHAR(100) NOT NULL,
    expected\_data\_type VARCHAR(50) NOT NULL,
    description TEXT NULL,
    is\_required BOOLEAN NOT NULL,
    created\_at TIMESTAMP NOT NULL,
    updated\_at TIMESTAMP NULL,
    UNIQUE(section\_id, variable\_name))}

    \item \texttt{template\_mappings(
    mapping\_id UUID PK NOT NULL,
    placeholder\_id UUID FK NOT NULL,
    entity\_id UUID FK NOT NULL,
    extraction\_rule TEXT NULL,
    is\_active BOOLEAN NOT NULL,
    UNIQUE(placeholder\_id, entity\_id))}

    \item \texttt{documents(
    document\_id UUID PK NOT NULL,
    workspace\_id UUID FK NOT NULL,
    user\_id UUID FK NOT NULL,
    title VARCHAR(255) NOT NULL,
    document\_type VARCHAR(50) NOT NULL,
    status VARCHAR(50) NOT NULL,
    created\_at TIMESTAMP NOT NULL,
    updated\_at TIMESTAMP NULL,
    deleted\_at TIMESTAMP NULL)}

    \item \texttt{document\_versions(
    version\_id UUID PK NOT NULL,
    document\_id UUID FK NOT NULL,
    version\_number INT NOT NULL,
    content TEXT NOT NULL,
    change\_summary TEXT NULL,
    status VARCHAR(50) NOT NULL,
    ticketed\_codes TEXT NULL,
    created\_at TIMESTAMP NOT NULL,
    updated\_at TIMESTAMP NULL,
    UNIQUE(document\_id, version\_number))}

    \item \texttt{change\_logs(
    log\_id UUID PK NOT NULL,
    version\_id UUID FK NOT NULL,
    change\_type VARCHAR(50) NOT NULL,
    change\_description TEXT NOT NULL,
    diff\_content TEXT NULL,
    created\_at TIMESTAMP NOT NULL,
    updated\_at TIMESTAMP NOT NULL)}

\end{itemize}
